\node[box, fill=white, draw=mtdark!35, align=left, text width=132mm, inner sep=8pt,
      anchor=north west] (eq) at (-1.0,0.35) {%
\small
\head{Symbol, air time and slot.} $\mathrm{SF}$, $\mathrm{BW}$, coding rate $4/d$;
$L$ = bytes on air (16-byte header included), $n_\mathrm{pre}=16$ symbols.
\[
  T_s=\frac{2^{\mathrm{SF}}}{\mathrm{BW}},\qquad
  R_b=\mathrm{SF}\cdot\frac{\mathrm{BW}}{2^{\mathrm{SF}}}\cdot\frac{4}{d}
      =\frac{\mathrm{SF}}{T_s}\cdot\frac{4}{d},\qquad
  \mathrm{ToA}(L)=N_\mathrm{sym}\,T_s
\]
\[
  N_\mathrm{sym}=\underbrace{(n_\mathrm{pre}+4.25)}_{\text{preamble}}+8+
  \max\!\left(\left\lceil\frac{8L-4\,\mathrm{SF}+28+16\,\mathrm{CRC}-20\,\mathrm{IH}}
  {4(\mathrm{SF}-2\,\mathrm{DE})}\right\rceil d,\;0\right),\qquad
  \mathrm{DE}=1\ \text{iff}\ T_s>16\,\mathrm{ms}
\]
\[
  \sigma=\max\!\left(2.25,\;N_\mathrm{CAD}+\tfrac12\right)T_s+T_\mathrm{oh},
  \qquad T_\mathrm{oh}=\underbrace{0.2}_{\text{propagation}}+\underbrace{0.4}_{\text{turnaround}}
  +\underbrace{7.0}_{\text{MAC}}=7.6\ \mathrm{ms}
\]
\head{Channel utilisation} --- six 10-second buckets, so a 60-second window, and it
counts \emph{everything} the radio spent on air: own transmissions plus every
reception, decodable or not.
\[
  U_{ch}=\frac{100}{60\,\mathrm{s}}\sum_j n_j\,\mathrm{ToA}_j\ [\%]
\]
\head{Contention window.} $w \equiv \mathrm{CW_{size}}$ is an \emph{exponent}, not a
number of slots: $w_{\min}=3$, $w_{\max}=8$. The firmware interpolates it linearly over a
fixed input range, in integer arithmetic:
\[
  w = w_{\min}+\left\lfloor (x-x_{\min})\,
      \frac{w_{\max}-w_{\min}}{x_{\max}-x_{\min}}\right\rfloor,
  \qquad
  x=\begin{cases}
      U_{ch}\ \text{over } [0,100]\ \% & \text{originating}\\[2pt]
      \mathrm{SNR}\ \text{over } [-20,+10]\ \mathrm{dB} & \text{relaying}
    \end{cases}
\]
\[
  \text{originating:}\quad w_\mathrm{orig}=\left\lfloor\frac{U_{ch}}{20}\right\rfloor+3,
  \qquad\qquad
  \text{relaying:}\quad w_\mathrm{relay}=\left\lfloor\frac{\mathrm{SNR}+20}{6}\right\rfloor+3
\]
\textbf{Nothing clamps the result}: an input above the top of its range gives
$w>w_{\max}$.

\head{The delay itself.} $\delta=k\,\sigma$, with $k$ drawn uniformly over integers:
\[
  \underbrace{k\sim\mathcal{U}\{0,\,2^{w}-1\}}_{\text{originating, any role}}
  \qquad
  \underbrace{k\sim\mathcal{U}\{0,\,2w-1\}}_{\substack{\text{relaying, \texttt{ROUTER}}\\ \text{(linear in }w\text{, not }2^w)}}
  \qquad
  \underbrace{k\sim\mathcal{U}\{2w_{\max},\;2w_{\max}+2^{w}-1\}}_{\text{relaying, every other role}}
\]
\[
  \delta_\mathrm{worst}=\left(2w_{\max}+2^{w}\right)\sigma,
  \qquad
  T_\mathrm{retx}=2\,\mathrm{ToA}+\left(2^{w}+2w_{\max}
    +2^{\lfloor (w_{\max}+w_{\min})/2\rfloor}\right)\sigma+4500\ \mathrm{ms}
\]
};

\node[note, text width=132mm, align=left, anchor=north west, text=mtdark!65] at (-1.0,-13.5)
  {Constants read from \texttt{meshtastic/firmware} \texttt{v2.7.26.54e0d8d}, the
   \texttt{firmware\_target} of the deployments: \texttt{CWmin}~3, \texttt{CWmax}~8,
   \texttt{SNR\_MIN}~$-20$, \texttt{SNR\_MAX}~$+10$, \texttt{NUM\_SYM\_CAD}~2, preamble

   \texttt{uint32} of milliseconds, so 17.84 ms is truncated to 17 and every $\delta$
