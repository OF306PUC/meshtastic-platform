% What a node does with a frame it just received: the rebroadcast decision.
% Build from this folder: pdflatex rebroadcast.tex  (font and colours: palette.tex).
%
% Firmware read at meshtastic/firmware tag v2.7.26.54e0d8d, 2026-10-27 — the
% firmware_target of the deployments. Sources: mesh/FloodingRouter.cpp,
% mesh/NextHopRouter.cpp (perhapsRebroadcast), mesh/PacketHistory.cpp,
% mesh/Router.cpp. The timing of the enqueued copy is the companion figure
% channel-access.tex.
\documentclass[tikz,border=6pt]{standalone}
\input{palette}

\tikzset{
  gate/.style={box, fill=white, draw=mtdark!60, align=center, minimum width=66mm},
  drop/.style={box, fill=mtdark!7, draw=mtdark!30, text=mtdark!80, font=\scriptsize,
               align=center, minimum width=40mm, minimum height=9mm},
  dup/.style={box, fill=lorablue!6, draw=lorablue!70, font=\scriptsize, align=left,
              text width=66mm, minimum height=0mm},
  act/.style={box, fill=mtgreen!16, draw=mtgreen7, align=center, minimum width=66mm},
  flow/.style={->, thick, draw=mtdark!75},
  alt/.style={->, thick, draw=mtdark!40, dashed},
  elab/.style={font=\scriptsize\itshape, inner sep=1.5pt, fill=white, text=mtdark!75},
}

\begin{document}
\begin{tikzpicture}[font=\small]

% ── the main column: every test a frame must pass to be relayed ───────────────
\node[radio, minimum width=66mm] (rx) at (0,0)
  {\textbf{LoRa frame received}\\[-1pt]
   \radiosub{\texttt{from} · \texttt{id} · \texttt{hop\_limit} · \texttt{hop\_start} · \texttt{next\_hop} · \texttt{rx\_snr}}};

\node[gate] (g1) at (0,-2.3)
  {\texttt{wasSeenRecently(from, id)}?\\[1pt]
   {\scriptsize a ring of $\geq 100$ (sender, id) records, oldest evicted ---}\\[-2pt]
   {\scriptsize no time expiry; our own sends are recorded too}};

\node[gate] (g2) at (0,-5.0) {\texttt{isToUs(p)} or \texttt{isFromUs(p)}?};

\node[gate] (g3) at (0,-7.0) {\texttt{hop\_limit > 0}?};

\node[gate] (g4) at (0,-9.2)
  {\texttt{isRebroadcaster()}?\\[1pt]
   {\scriptsize role $\neq$ \texttt{CLIENT\_MUTE} \textbf{and} \texttt{rebroadcast\_mode} $\neq$ \texttt{NONE}}};

\node[gate] (g5) at (0,-11.6)
  {\texttt{next\_hop == 0} (flood)\\[-1pt] or \texttt{next\_hop == me}?};

\node[act] (act) at (0,-14.1)
  {\textbf{relay it}\\[1pt]
   {\scriptsize copy the packet · \texttt{hop\_limit{-}{-}} · \texttt{relay\_node $\leftarrow$ me}}\\[-2pt]
   {\scriptsize \texttt{txQueue.enqueue()} --- nothing is sent yet}};

\node[other, minimum width=66mm, font=\small] (out) at (0,-16.3)
  {\textbf{channel access}: wait $\delta_\mathrm{relay}$, then CAD\\[-1pt]
   {\scriptsize see \texttt{channel-access} figure}};

\draw[flow] (rx) -- (g1);
\draw[flow] (g1) -- node[elab, right] {new} (g2);
\draw[flow] (g2) -- node[elab, right] {someone else's} (g3);
\draw[flow] (g3) -- node[elab, right] {hops left} (g4);
\draw[flow] (g4) -- node[elab, right] {yes} (g5);
\draw[flow] (g5) -- node[elab, right] {we are on the path} (act);
\draw[flow] (act) -- (out);

% ── the left column: where a frame dies ──────────────────────────────────────
\node[drop] (d2) at (-5.7,-5.0) {ours: hand to the modules\\ and the phone — no relay};
\node[drop] (d3) at (-5.7,-7.0) {hop budget spent --- drop\\ {\itshape one relay is all a packet}\\ {\itshape originated at \texttt{hop\_limit 1} gets}};
\node[drop] (d4) at (-5.7,-9.2) {muted, or rebroadcasting\\ switched off — drop};
\node[drop] (d5) at (-5.7,-11.6) {another node is the\\ chosen next hop — drop};

\draw[alt] (g2.west) -- node[elab, above] {ours} (d2.east);
\draw[alt] (g3.west) -- node[elab, above] {\texttt{== 0}} (d3.east);
\draw[alt] (g4.west) -- node[elab, above] {no} (d4.east);
\draw[alt] (g5.west) -- node[elab, above] {no} (d5.east);

% ── the right column: the duplicate path, where most copies end ──────────────
\node[dup] (u1) at (7.1,-1.3)
  {\textbf{a better copy?} \texttt{hop\_limit} higher than the one already queued
   $\Rightarrow$ \texttt{removePendingTXPacket()} and queue this one instead.};
\node[dup] (u2) at (7.1,-3.7)
  {\textbf{the sender retrying?} \texttt{hop\_start == hop\_limit} and the packet is no
   longer in \texttt{txQueue} $\Rightarrow$ relay once more, so the source gets an
   implicit ACK.};
\node[dup] (u3) at (7.1,-6.9)
  {\textbf{otherwise} \texttt{perhapsCancelDupe()}:\\[2pt]
   \texttt{CLIENT} — \textbf{cancel} its own queued relay: a neighbour already
   covered this packet. This is the mechanism the SNR-weighted delay of the
   companion figure exists to drive.\\[2pt]
   \texttt{ROUTER}, \texttt{ROUTER\_LATE} — never cancel. \texttt{ROUTER\_LATE}
   re-arms the queued copy at $\delta_\mathrm{worst}$ instead.};
\node[drop, minimum width=66mm] (dupend) at (7.1,-9.6) {duplicate — stop here\\ ($\mathtt{rxDupe}{+}{+}$)};

\draw[flow] (g1.east) -| node[elab, above, pos=0.22] {seen} (u1.south);
\draw[flow] (u1) -- node[elab, right] {no} (u2);
\draw[flow] (u2) -- node[elab, right] {no} (u3);
\draw[flow] (u3) -- (dupend);

\begin{scope}[on background layer]
  \node[draw=lorablue!60, dashed, rounded corners=6pt, fill=lorablue!3,
        inner sep=8pt, fit=(u1)(u2)(u3)(dupend)] {};
\end{scope}

% ── what is NOT in this figure ───────────────────────────────────────────────
\node[note, text width=152mm, align=left, anchor=north west, text=mtdark!75] at (-7.9,-17.8)
  {\textbf{Earlier gates, not shown.} \texttt{LOCAL\_ONLY} and \texttt{KNOWN\_ONLY}
   drop at decode time (unknown sender / foreign channel), and
   \texttt{CORE\_PORTNUMS\_ONLY} cancels after decode on a non-core portnum —
   all three reach \texttt{isRebroadcaster()} as a frame already thrown away.};

\end{tikzpicture}
\end{document}
