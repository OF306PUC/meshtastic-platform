\node[note, text width=58mm, align=left, anchor=north west, text=mtdark!80] at (-8.9,-11.2)
  {\textbf{Not 802.11.} The backoff is \emph{re-drawn}, never frozen and resumed,
   and it never doubles after a loss: there are no ACKs at this layer, so the
   window is a function of $U_{ch}$ (own traffic) or \texttt{rx\_snr} (a relay)
   alone. CAD cannot separate two stations that drew the same $k$ --- that tie
   is the collision mechanism. A re-draw is capped at
   $t_\mathrm{tx} \leq t_\mathrm{now} + 2\,\delta_\mathrm{worst}$.};

\node[box, fill=white, draw=mtdark!35, align=left, text width=132mm, inner sep=8pt,
      anchor=north west] (eq) at (-1.0,0.35) {%
\small
\head{Symbol, air time and slot.} $\mathrm{SF}$, $\mathrm{BW}$, coding rate $4/d$;
$L$ = bytes on air (16-byte header included), $n_\mathrm{pre}=16$ symbols.
\[
  T_s=\frac{2^{\mathrm{SF}}}{\mathrm{BW}},\qquad
  R_b=\mathrm{SF}\cdot\frac{\mathrm{BW}}{2^{\mathrm{SF}}}\cdot\frac{4}{d}
      =\frac{\mathrm{SF}}{T_s}\cdot\frac{4}{d},\qquad
  \mathrm{ToA}(L)=N_\mathrm{sym}\,T_s
\]
\[
  N_\mathrm{sym}=\underbrace{(n_\mathrm{pre}+4.25)}_{\text{preamble}}+8+
  \max\!\left(\left\lceil\frac{8L-4\,\mathrm{SF}+28+16\,\mathrm{CRC}-20\,\mathrm{IH}}
  {4(\mathrm{SF}-2\,\mathrm{DE})}\right\rceil d,\;0\right),\qquad
  \mathrm{DE}=1\ \text{iff}\ T_s>16\,\mathrm{ms}
\]
\[
  \sigma=\max\!\left(2.25,\;N_\mathrm{CAD}+\tfrac12\right)T_s+T_\mathrm{oh},
  \qquad T_\mathrm{oh}=\underbrace{0.2}_{\text{propagation}}+\underbrace{0.4}_{\text{turnaround}}
  +\underbrace{7.0}_{\text{MAC}}=7.6\ \mathrm{ms}
\]
\head{Channel utilisation} --- six 10-second buckets, so a 60-second window, and it
counts \emph{everything} the radio spent on air: own transmissions plus every
reception, decodable or not.
\[
  U_{ch}=\frac{100}{60\,\mathrm{s}}\sum_j n_j\,\mathrm{ToA}_j\ [\%]
\]
\head{Contention window.} $w \equiv \mathrm{CW_{size}}$ is an \emph{exponent};
$w_{\min}=3$, $w_{\max}=8$. Both maps are Arduino \texttt{map()}: integer, and
\textbf{not clamped} at either end.
\[
  \underbrace{w_\mathrm{orig}=\mathrm{map}(U_{ch};0,100\to w_{\min},w_{\max})
     =\left\lfloor\frac{U_{ch}}{20}\right\rfloor+3}_{\text{originating}}
  \qquad
  \underbrace{w_\mathrm{relay}=\mathrm{map}(\mathrm{SNR};-20,10\to w_{\min},w_{\max})
     =\left\lfloor\frac{\mathrm{SNR}+20}{6}\right\rfloor+3}_{\text{relaying}}
\]
\head{The delay itself.} $\delta=k\,\sigma$, with $k$ drawn uniformly over integers:
\[
  \underbrace{k\sim\mathcal{U}\{0,\,2^{w}-1\}}_{\text{originating, any role}}
  \qquad
  \underbrace{k\sim\mathcal{U}\{0,\,2w-1\}}_{\substack{\text{relaying, \texttt{ROUTER}}\\ \text{(linear in }w\text{, not }2^w)}}
  \qquad
  \underbrace{k\sim\mathcal{U}\{2w_{\max},\;2w_{\max}+2^{w}-1\}}_{\text{relaying, every other role}}
\]
\[
  \delta_\mathrm{worst}=\left(2w_{\max}+2^{w}\right)\sigma,
  \qquad
  T_\mathrm{retx}=2\,\mathrm{ToA}+\left(2^{w}+2w_{\max}
    +2^{\lfloor (w_{\max}+w_{\min})/2\rfloor}\right)\sigma+4500\ \mathrm{ms}
\]
};

\node[box, fill=amber!8, draw=amber!70, inner sep=7pt, align=left, text width=62mm,
      anchor=north west, font=\scriptsize] at (10.9,-18.3)
 {\textbf{The ordering is anticorrelated with link quality.} A node that heard the
  packet \emph{badly} --- the far one --- draws from a short window and relays first;
  its well-connected neighbours hear that relay and cancel their own
  (\texttt{perhapsCancelDupe}). The relay population is therefore biased towards
  weak links, and any SNR distribution built from $\mathtt{hop}>0$ packets inherits
  that bias.\\[3pt]
  \textbf{Neither map is clamped.} $U_{ch}>100\%$ or $\mathrm{SNR}>+10$ dB gives
  $w>w_{\max}$; LoRa SNR saturates near $+13$ dB, so $w\leq 8$ in practice.};
